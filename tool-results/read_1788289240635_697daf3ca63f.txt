    19→    19→function MiniBar({ data }: { data: { label: string; value: number }[] }) {
    20→    20→  const max = Math.max(1, ...data.map((d) => d.value));
    21→    21→  return (
    22→    22→    <div className="flex h-36 items-end gap-1" role="img" aria-label="Bar chart">
    23→    23→      {data.map((d, i) => (
    24→    24→        <div key={i} className="group relative flex-1">
    25→    25→          <div
    26→    26→            className={cn("w-full rounded-t transition-all", d.value > 0 ? "bg-primary/80 group-hover:bg-primary" : "bg-muted")}
    27→    27→            style={{ height: `${Math.max(3, (d.value / max) * 130)}px` }}
    28→    28→          />
    29→    29→          <span className="pointer-events-none absolute -top-7 left-1/2 hidden -translate-x-1/2 whitespace-nowrap rounded bg-foreground px-1.5 py-0.5 text-[10px] text-background group-hover:block">
    30→    30→            {d.label}: {d.value}
    31→    31→          </span>
    32→    32→        </div>
    33→    33→      ))}
    34→    34→    </div>
    35→    35→  );
    36→    36→}

    58→    58→
    59→    59→function timeAgo(iso: string): string {
    60→    60→  const seconds = Math.floor((Date.now() - new Date(iso).getTime()) / 1000);
    61→    61→  if (seconds < 60) return "just now";
    62→    62→  const minutes = Math.floor(seconds / 60);
    63→    63→  if (minutes < 60) return `${minutes}m ago`;
    64→    64→  const hours = Math.floor(minutes / 60);
    65→    65→  if (hours < 24) return `${hours}h ago`;
    66→    66→  const days = Math.floor(hours / 24);
    67→    67→  if (days < 30) return `${days}d ago`;
    68→    68→  return new Date(iso).toLocaleDateString("en-GB");
    69→    69→}
    70→    70→
    71→    71→function formatBytes(bytes: number): string {
    72→    72→  if (bytes < 1024) return `${bytes} B`;
    73→    73→  if (bytes < 1048576) return `${(bytes / 1024).toFixed(1)} KB`;
    74→    74→  return `${(bytes / 1048576).toFixed(1)} MB`;
    75→    75→}
